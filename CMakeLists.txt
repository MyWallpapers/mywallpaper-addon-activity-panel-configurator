cmake_minimum_required(VERSION 3.24)
project(MyWallpaperActivityPanel VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_SUPPRESS_REGENERATION ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/manifest.json" MW_MANIFEST)
configure_file(include/manifest_data.hpp.in manifest_data.hpp @ONLY)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/contracts/audio-system/v1/schema.json" A18_SCHEMA)
# MSVC limits each source literal. Adjacent line-sized raw literals preserve
# the exact pinned schema bytes while staying below that compiler limit.
string(REPLACE "\n" "\n)A18schema\"\nR\"A18schema(" A18_SCHEMA "${A18_SCHEMA}")
configure_file(include/audio_schema.hpp.in audio_schema.hpp @ONLY)
add_library(activity_model STATIC src/model.cpp src/wire.cpp src/audio_contract.cpp src/audio_service.cpp)
target_include_directories(activity_model PUBLIC include third_party "${CMAKE_CURRENT_BINARY_DIR}")
if(MSVC)
  target_compile_options(activity_model PUBLIC /utf-8 /W4 /permissive- /EHsc /Brepro)
else()
  target_compile_options(activity_model PUBLIC -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_executable(contract_tests tests/contract_tests.cpp)
target_link_libraries(contract_tests PRIVATE activity_model)
add_test(NAME activity_contracts COMMAND contract_tests)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/contracts/audio-system/v1/examples.json" A18_EXAMPLES)
configure_file(tests/audio_examples.hpp.in audio_examples.hpp @ONLY)
add_executable(audio_service_tests tests/audio_service_tests.cpp)
target_link_libraries(audio_service_tests PRIVATE activity_model)
add_test(NAME activity_audio_service COMMAND audio_service_tests)
if(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Only Windows x64 is supported")
  endif()
  if(NOT MSVC)
    message(FATAL_ERROR "Windows candidates require MSVC")
  endif()
  add_executable(companion src/main.cpp src/panel.cpp src/accessibility.cpp)
  target_link_libraries(companion PRIVATE activity_model gdiplus user32 gdi32 ole32 oleaut32 oleacc shcore uuid)
  target_compile_definitions(companion PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  add_test(NAME activity_owned_window COMMAND companion --self-test "${CMAKE_CURRENT_BINARY_DIR}/fixture-output")
  set_tests_properties(activity_owned_window PROPERTIES TIMEOUT 30)
  add_executable(transport_tests tests/windows_transport.cpp)
  target_link_libraries(transport_tests PRIVATE activity_model user32)
  target_compile_definitions(transport_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  add_test(NAME activity_private_transport COMMAND transport_tests "$<TARGET_FILE:companion>" "${CMAKE_CURRENT_BINARY_DIR}/transport-output")
  set_tests_properties(activity_private_transport PROPERTIES TIMEOUT 30)
  add_test(NAME activity_window_policy COMMAND companion --window-policy-test)
  set_tests_properties(activity_window_policy PROPERTIES TIMEOUT 15)
endif()
if(MW_NATIVE_DEVELOPMENT_MODULE)
  include("${MW_NATIVE_DEVELOPMENT_MODULE}")
  if(WIN32)
    mw_native_development(UNIT_TARGETS contract_tests audio_service_tests UNIT_TESTS activity_contracts activity_audio_service
      RUNTIME_TARGETS companion FIXTURE_TARGETS companion transport_tests FIXTURE_TESTS activity_owned_window activity_private_transport activity_window_policy)
  else()
    mw_native_development(UNIT_TARGETS contract_tests audio_service_tests UNIT_TESTS activity_contracts activity_audio_service)
  endif()
endif()
